```python
import bpy
import bmesh
import mathutils
from math import radians

# Palette
RED = (0.82, 0.18, 0.13, 1.0)
BROWN = (0.55, 0.35, 0.17, 1.0)
D_GREY = (0.20, 0.20, 0.22, 1.0)
N_BLACK = (0.08, 0.08, 0.09, 1.0)
L_GREY = (0.60, 0.60, 0.63, 1.0)
CREAM = (0.96, 0.95, 0.90, 1.0)
SKY = (0.66, 0.85, 0.91, 1.0)
WARM = (1.00, 0.95, 0.80, 1.0)

def create_mat(name, color, rough=0.5):
    mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = color
    bsdf.inputs["Roughness"].default_value = rough
    return mat

def create_part(name, mesh_func, mat, loc=(0,0,0)):
    me = bpy.data.meshes.new(name)
    bm = bmesh.new()
    mesh_func(bm)
    bm.to_mesh(me)
    bm.free()
    ob = bpy.data.objects.new(name, me)
    ob.location = loc
    bpy.context.collection.objects.link(ob)
    ob.data.materials.append(mat)
    return ob

def build_wheel(name, radius, width, loc):
    # Wheel object at hub center, local X is axle
    me = bpy.data.meshes.new(name)
    bm = bmesh.new()
    # Tyre
    bmesh.ops.create_cone(bm, cap_ends=True, cap_tris=False, segments=24, radius1=radius, radius2=radius, depth=width)
    # Hub
    bmesh.ops.create_cone(bm, cap_ends=True, cap_tris=False, segments=16, radius1=radius*0.6, radius2=radius*0.6, depth=width*1.1)
    
    # Rotate mesh data so X is axle (cylinder created along Z by default)
    mat_rot = mathutils.Matrix.Rotation(radians(90), 4, 'Y')
    bmesh.ops.rotate(bm, cent=(0,0,0), matrix=mat_rot, verts=bm.verts)
    
    bm.to_mesh(me)
    bm.free()
    ob = bpy.data.objects.new(name, me)
    ob.location = loc
    bpy.context.collection.objects.link(ob)
    ob.data.materials.append(create_mat("Mat_Tyre", N_BLACK, 0.9))
    ob.data.materials.append(create_mat("Mat_Hub", L_GREY, 0.4))
    # Assign hub mat to inner cone
    for poly in ob.data.polygons:
        if poly.index > 48: poly.material_index = 1
    return ob

# Materials
m_red = create_mat("Red", RED, 0.4)
m_grey = create_mat("DGrey", D_GREY, 0.8)
m_sky = create_mat("Sky", SKY, 0.15)
m_cream = create_mat("Cream", CREAM, 0.7)
m_warm = create_mat("Warm", WARM, 0.3)

# Build Parts
# Bonnet
create_part("Bonnet", lambda bm: bmesh.ops.create_cube(bm, size=1.0), m_red).scale = (1.2, 0.8, 1.4)
# Cabin
create_part("Cabin", lambda bm: bmesh.ops.create_cube(bm, size=1.0), m_red, (0, 0.8, -0.5)).scale = (1.1, 1.2, 1.0)
# Windows
create_part("Window", lambda bm: bmesh.ops.create_cube(bm, size=1.0), m_sky, (0, 0.8, -0.45)).scale = (1.0, 1.0, 0.9)
# Plate
create_part("Plate", lambda bm: bmesh.ops.create_cube(bm, size=1.0), m_cream, (0, 1.1, -1.3)).scale = (0.5, 0.22, 0.05)

# Wheels
build_wheel("Wheel_FL", 0.32, 0.25, (1.1, 0.32, 0.8))
build_wheel("Wheel_FR", 0.32, 0.25, (-1.1, 0.32, 0.8))
build_wheel("Wheel_RL", 0.55, 0.35, (1.1, 0.55, -0.8))
build_wheel("Wheel_RR", 0.55, 0.35, (-1.1, 0.55, -0.8))

# Exhaust
exh = create_part("Exhaust", lambda bm: bmesh.ops.create_cone(bm, cap_ends=True, cap_tris=False, segments=8, radius1=0.1, radius2=0.1, depth=1.0), m_grey, (-0.6, 1.2, 0.2))
exh.rotation_euler = (0.1, 0, 0)

bpy.ops.export_scene.gltf(filepath="__GLB_OUT__", export_format='GLB')
print("ASSET_BUILT")
print("DIMS Vector((2.2, 1.9, 2.6))")
```
