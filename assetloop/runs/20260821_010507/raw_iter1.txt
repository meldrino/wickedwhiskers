import bpy
import bmesh
import mathutils
from math import radians

# Palette
RED = (0.82, 0.18, 0.13, 1.0)
BROWN = (0.55, 0.35, 0.17, 1.0)
DARK_GREY = (0.20, 0.20, 0.22, 1.0)
NEAR_BLACK = (0.08, 0.08, 0.09, 1.0)
LIGHT_GREY = (0.60, 0.60, 0.63, 1.0)
CREAM = (0.96, 0.95, 0.90, 1.0)
SKY = (0.66, 0.85, 0.91, 1.0)
WARM = (1.00, 0.95, 0.80, 1.0)

def create_mat(name, color, roughness=0.5):
    mat = bpy.data.materials.new(name)
    mat.use_nodes = True
    bsdf = mat.node_tree.nodes["Principled BSDF"]
    bsdf.inputs["Base Color"].default_value = color
    bsdf.inputs["Roughness"].default_value = roughness
    return mat

def create_part(name, mesh_data, mat, pos=(0,0,0)):
    ob = bpy.data.objects.new(name, mesh_data)
    ob.location = pos
    bpy.context.collection.objects.link(ob)
    ob.data.materials.append(mat)
    return ob

# Materials
m_red = create_mat("Red", RED, 0.4)
m_dark = create_mat("Dark", DARK_GREY, 0.6)
m_black = create_mat("Black", NEAR_BLACK, 0.6)
m_light = create_mat("Light", LIGHT_GREY, 0.4)
m_cream = create_mat("Cream", CREAM, 0.8)
m_sky = create_mat("Sky", SKY, 0.15)
m_warm = create_mat("Warm", WARM, 0.3)

# Helper for wheel geometry
def build_wheel(name, radius, width, pos):
    bm = bmesh.new()
    bmesh.ops.create_cone(bm, cap_ends=True, cap_tris=False, segments=24, radius1=radius, radius2=radius, depth=width)
    me = bpy.data.meshes.new(name)
    bm.to_mesh(me)
    bm.free()
    # Rotate mesh data so local X is the axle
    me.transform(mathutils.Matrix.Rotation(radians(90), 4, 'Z'))
    return create_part(name, me, m_black, pos)

# Build Parts
# Body (Bonnet)
bm = bmesh.new()
bmesh.ops.create_cube(bm, size=1.0)
me = bpy.data.meshes.new("Bonnet")
bm.to_mesh(me)
me.transform(mathutils.Matrix.Scale(1.2, 4, (1,0,0)) @ mathutils.Matrix.Scale(0.8, 4, (0,1,0)) @ mathutils.Matrix.Scale(1.5, 4, (0,0,1)))
create_part("Bonnet", me, m_red, (0, 0.8, 0.75))

# Cabin
bm = bmesh.new()
bmesh.ops.create_cube(bm, size=1.0)
me = bpy.data.meshes.new("Cabin")
bm.to_mesh(me)
me.transform(mathutils.Matrix.Scale(1.4, 4, (1,0,0)) @ mathutils.Matrix.Scale(1.2, 4, (0,1,0)) @ mathutils.Matrix.Scale(1.0, 4, (0,0,1)))
create_part("Cabin", me, m_red, (0, 1.2, -0.6))

# Wheels
w_fl = build_wheel("Wheel_FL", 0.32, 0.25, (1.1, 0.32, 0.8))
w_fr = build_wheel("Wheel_FR", 0.32, 0.25, (-1.1, 0.32, 0.8))
w_rl = build_wheel("Wheel_RL", 0.55, 0.35, (1.1, 0.55, -0.8))
w_rr = build_wheel("Wheel_RR", 0.55, 0.35, (-1.1, 0.55, -0.8))

# Number Plate
bm = bmesh.new()
bmesh.ops.create_cube(bm, size=1.0)
me = bpy.data.meshes.new("Plate")
bm.to_mesh(me)
me.transform(mathutils.Matrix.Scale(0.5, 4, (1,0,0)) @ mathutils.Matrix.Scale(0.22, 4, (0,1,0)) @ mathutils.Matrix.Scale(0.05, 4, (0,0,1)))
create_part("Plate", me, m_cream, (0, 1.1, -1.3))

bpy.ops.export_scene.gltf(filepath="__GLB_OUT__", export_format='GLB')
