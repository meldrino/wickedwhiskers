def create_box_object(name, size, location, mat_name, rotation=(0, 0, 0)):
    """Creates a box mesh with geometry transformed in mesh space."""
    me = bpy.data.meshes.new(name)
    bm = bmesh.new()
    bmesh.ops.create_cube(bm, size=1.0)
    
    # Scale cube to desired size
    for v in bm.verts:
        v.co.x *= size[0]
        v.co.y *= size[1]
        v.co.z *= size[2]
        
    bm.to_mesh(me)
    bm.free()
    
    # Apply rotation and translation to mesh data directly
    rot_mat = Matrix.Rotation(rotation[0], 4, 'X') @ \
              Matrix.Rotation(rotation[1], 4, 'Y') @ \
              Matrix.Rotation(rotation[2], 4, 'Z')
    trans_mat = Matrix.Translation(Vector(location))
    me.transform(trans_mat @ rot_mat)
    
    ob = bpy.data.objects.new(name, me)
    ob.data.materials.append(materials[mat_name])
    bpy.context.collection.objects.link(ob)
    return ob

def create_cylinder_object(name, radius, depth, location, mat_name, axis='Y', segments=20):
    """Creates a cylinder aligned with axis ('X', 'Y', or 'Z')."""
    me = bpy.data.meshes.new(name)
    bm = bmesh.new()
    bmesh.ops.create_cone(bm, cap_ends=True, cap_tris=False, segments=segments,
                          radius1=radius, radius2=radius, depth=depth)
    
    # Cone default axis is Z; rotate to X or Y if needed
    if axis == 'X':
        rot = Matrix.Rotation(math.radians(90), 4, 'Y')
        for v in bm.verts:
            v.co = rot @ v.co
    elif axis == 'Y':
        rot = Matrix.Rotation(math.radians(-90), 4, 'X')
        for v in bm.verts:
            v.co = rot @ v.co

    bm.to_mesh(me)
    bm.free()
    
    me.transform(Matrix.Translation(Vector(location)))
    ob = bpy.data.objects.new(name, me)
    ob.data.materials.append(materials[mat_name])
    bpy.context.collection.objects.link(ob)
    return ob

def create_sphere_object(name, radius, location, mat_name):

def create_mudguard_arch(name, center, inner_radius, width, thickness, mat_name):
    """Creates an arched mudguard over rear wheels."""
    me = bpy.data.meshes.new(name)
    bm = bmesh.new()
    
    segments = 8
    start_angle = math.radians(15)
    end_angle = math.radians(165)
    
    verts_outer = []
    verts_inner = []
    
    for i in range(segments + 1):
        t = i / segments
        angle = start_angle + t * (end_angle - start_angle)
        
        # Local arch in Y-Z plane
        cos_a = math.cos(angle)
        sin_a = math.sin(angle)
        
        r_in = inner_radius
        r_out = inner_radius + thickness
        
        # Left and Right edges along X
        hx = width / 2.0
        v0 = bm.verts.new((-hx, r_in * sin_a, r_in * cos_a))
        v1 = bm.verts.new((hx, r_in * sin_a, r_in * cos_a))
        v2 = bm.verts.new((hx, r_out * sin_a, r_out * cos_a))
        v3 = bm.verts.new((-hx, r_out * sin_a, r_out * cos_a))
        
        verts_outer.append((v0, v1, v2, v3))
        
    for i in range(segments):
        curr_v = verts_outer[i]
        next_v = verts_outer[i+1]
        # Inner face
        bm.faces.new((curr_v[0], next_v[0], next_v[1], curr_v[1]))
        # Outer face
        bm.faces.new((curr_v[3], curr_v[2], next_v[2], next_v[3]))
        # Side face +X
        bm.faces.new((curr_v[1], next_v[1], next_v[2], curr_v[2]))
        # Side face -X
        bm.faces.new((curr_v[0], curr_v[3], next_v[3], next_v[0]))
        
    # Caps at ends
    bm.faces.new((verts_outer[0][0], verts_outer[0][1], verts_outer[0][2], verts_outer[0][3]))
    bm.faces.new((verts_outer[-1][3], verts_outer[-1][2], verts_outer[-1][1], verts_outer[-1][0]))
    
    bm.to_mesh(me)
    bm.free()
    
    me.transform(Matrix.Translation(Vector(center)))
    ob = bpy.data.objects.new(name, me)
    ob.data.materials.append(materials[mat_name])
    bpy.context.collection.objects.link(ob)
    return ob

# 4 wheels named exactly Wheel_FL, Wheel_FR, Wheel_RL, Wheel_RR
# Tyres along X axis. Object origin at hub center.
def create_wheel(name, hub_location, radius, width, hub_radius):
    me = bpy.data.meshes.new(name)
    bm = bmesh.new()
    
    # Tyre main body
    res_tyre = bmesh.ops.create_cone(bm, cap_ends=True, cap_tris=False, segments=24,
                                      radius1=radius, radius2=radius, depth=width)
    for f in res_tyre['faces']:
        f.material_index = 0
        
    # Tread ridge in center
    res_tread = bmesh.ops.create_cone(bm, cap_ends=True, cap_tris=False, segments=24,
                                       radius1=radius * 1.035, radius2=radius * 1.035, depth=width * 0.35)
    for f in res_tread['faces']:
        f.material_index = 0
        
    # Hub cap cylinder (slightly wider than tyre center so it's clearly readable)
    res_hub = bmesh.ops.create_cone(bm, cap_ends=True, cap_tris=False, segments=20,
                                     radius1=hub_radius, radius2=hub_radius, depth=width * 1.05)
    for f in res_hub['faces']:
        f.material_index = 1
        
    # Rotate mesh geometry so cylinder axis is along X (local axle)
    rot = Matrix.Rotation(math.radians(90), 4, 'Y')
    for v in bm.verts:
        v.co = rot @ v.co
        
    bm.to_mesh(me)
    bm.free()
    
    ob = bpy.data.objects.new(name, me)
    ob.location = Vector(hub_location) # Origin sits directly at hub center
    
    # Material slot 0: Near_Black, slot 1: Light_Grey
    ob.data.materials.append(materials["Near_Black"])
    ob.data.materials.append(materials["Light_Grey"])
    
    bpy.context.collection.objects.link(ob)
    return ob

# Compute overall scene bounding box across all mesh objects

        for corner in obj.bound_box:

            min_coords.y = min(min_coords.y, world_corner.y)
            min_coords.z = min(min_coords.z, world_corner.z)
            max_coords.x = max(max_coords.x, world_corner.x)

            max_coords.z = max(max_coords.z, world_corner.z)
