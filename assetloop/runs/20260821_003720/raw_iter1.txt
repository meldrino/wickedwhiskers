# -----------------------------------------------------------------------------
# MATERIALS (Wicked Whiskers Game Palette)
# -----------------------------------------------------------------------------
def get_material(name, color, roughness=0.5):
    mat = bpy.data.materials.get(name)
    if mat is None:
        mat = bpy.data.materials.new(name)
        mat.use_nodes = True
        bsdf = mat.node_tree.nodes.get("Principled BSDF")
        if bsdf:
            bsdf.inputs["Base Color"].default_value = (color[0], color[1], color[2], 1.0)
            bsdf.inputs["Roughness"].default_value = roughness
    return mat

mat_red         = get_material("TractorRed",   (0.82, 0.18, 0.13), 0.40)

# MESH CREATION HELPERS
# -----------------------------------------------------------------------------
def create_box(name, size, loc, mat, parent=None):
    me = bpy.data.meshes.new(name)
    bm = bmesh.new()
    bmesh.ops.create_cube(bm, size=1.0)
    # Scale cube to desired dimensions
    scale_mat = Matrix.Diagonal((size[0], size[1], size[2], 1.0))
    bmesh.ops.transform(bm, matrix=scale_mat, verts=bm.verts)
    bm.to_mesh(me)
    bm.free()
    
    ob = bpy.data.objects.new(name, me)
    ob.location = Vector(loc)
    ob.data.materials.append(mat)
    bpy.context.collection.objects.link(ob)
    return ob

def create_cylinder(name, radius, depth, loc, rot=(0, 0, 0), mat=mat_dark_grey, segments=24):
    me = bpy.data.meshes.new(name)
    bm = bmesh.new()
    bmesh.ops.create_cone(
        bm,
        cap_ends=True,
        cap_tris=False,
        segments=segments,
        radius1=radius,
        radius2=radius,
        depth=depth
    )
    # Cylinder default axis is Z; apply internal rotation if provided
    rot_mat = Euler(rot, 'XYZ').to_matrix().to_4x4()
    bmesh.ops.transform(bm, matrix=rot_mat, verts=bm.verts)
    bm.to_mesh(me)
    bm.free()
    
    ob = bpy.data.objects.new(name, me)
    ob.location = Vector(loc)
    ob.data.materials.append(mat)
    bpy.context.collection.objects.link(ob)
    return ob

# -----------------------------------------------------------------------------
# 1. WHEEL BUILDER (FL, FR, RL, RR)
# Object origin is exactly at hub center; spins smoothly around local X axis.
# -----------------------------------------------------------------------------
def build_wheel(name, radius, width, loc, is_left=True):
    me = bpy.data.meshes.new(name)
    bm = bmesh.new()

    # Materials needed on wheel
    me.materials.append(mat_near_black)  # index 0
    me.materials.append(mat_light_grey)  # index 1
    me.materials.append(mat_dark_grey)   # index 2

    # Rotate Z-axis cylinder to lie along X-axis
    rot_to_x = Euler((0, math.radians(90), 0), 'XYZ').to_matrix().to_4x4()

    # 1. Main tyre body
    ret_tyre = bmesh.ops.create_cone(
        bm, cap_ends=True, cap_tris=False, segments=28,
        radius1=radius, radius2=radius, depth=width
    )
    bmesh.ops.transform(bm, matrix=rot_to_x, verts=ret_tyre['verts'])
    for f in ret_tyre['faces']:
        f.material_index = 0

    # 2. Tyre central tread rib (chunky toy detail)
    ret_tread = bmesh.ops.create_cone(
        bm, cap_ends=True, cap_tris=False, segments=28,
        radius1=radius * 1.035, radius2=radius * 1.035, depth=width * 0.42
    )
    bmesh.ops.transform(bm, matrix=rot_to_x, verts=ret_tread['verts'])
    for f in ret_tread['faces']:
        f.material_index = 0

    # 3. Light grey Hub cap (slightly inset/protruding on outer face)
    hub_radius = radius * 0.58
    hub_width = width * 1.04
    ret_hub = bmesh.ops.create_cone(
        bm, cap_ends=True, cap_tris=False, segments=24,
        radius1=hub_radius, radius2=hub_radius, depth=hub_width
    )
    bmesh.ops.transform(bm, matrix=rot_to_x, verts=ret_hub['verts'])
    for f in ret_hub['faces']:
        f.material_index = 1

    # 4. Central axle nut / lug detail
    lug_offset_x = (width * 0.53) if not is_left else (-width * 0.53)
    ret_lug = bmesh.ops.create_cone(
        bm, cap_ends=True, cap_tris=False, segments=12,
        radius1=hub_radius * 0.35, radius2=hub_radius * 0.35, depth=width * 0.16
    )
    trans_lug = Matrix.Translation((0, 0, lug_offset_x))
    bmesh.ops.transform(bm, matrix=trans_lug, verts=ret_lug['verts'])
    bmesh.ops.transform(bm, matrix=rot_to_x, verts=ret_lug['verts'])
    for f in ret_lug['faces']:
        f.material_index = 2

    bm.to_mesh(me)
    bm.free()

    ob = bpy.data.objects.new(name, me)
    ob.location = Vector(loc)
    bpy.context.collection.objects.link(ob)
    return ob
